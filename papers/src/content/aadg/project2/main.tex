\begin{frame}[fragile]
  \frametitle{Classifier algorithm description (training)}
  \begin{enumerate}
    \scriptsize
    \item Build signatures on per-class basis
    \begin{enumerate}
      \scriptsize
      \item For each dataset associated with the class we open the gzip archive
      \item We concatenate each sequence into a buffer and append $N$ fragment of size $KMER\_LEN$ each time
      \item If we have enough sequences i.e loaded $DATASET\_CHUNK\_SIZE = 60000$ reads, then:
      \begin{enumerate}
        \scriptsize
        \item We generate kmers of length $KMER\_LEN = 20$ using simple binary shifts (lexicographic ordering)
        \item We sort and get unique kmer values along number of occurences
        \item We get $sig\_per\_step = 70000$ minimums and the same number of maximum kmer values
        \item We save that into two dictionaries $kmer -> occurences\_no$
      \end{enumerate}
      \item If we loaded at least $200000$ reads since the last time we also do the following:
      \begin{enumerate}
        \scriptsize
        \item Convert the dictionary into a numpy array of $np.int_{64}$ and merge it into single document signature
      \end{enumerate}
    \end{enumerate}
    \item Load test cases
  \end{enumerate}
\end{frame}

\begin{frame}[fragile]
  \frametitle{Classifier algorithm description (evaluation)}
  \begin{enumerate}
    \scriptsize
    \item For each dataset we do the same thing as for the classes, but all the chunk sizes are $x10$ greater
    \item Classify sequences i.e for each class do the following:
    \begin{enumerate}
      \scriptsize
      \item Merge signature of test dataset and class together
      \item Sort it (ascending)
      \item Split it in half using bit flags (discussed later) into ascending lists of min and max kmers occurences
      \item For each of two such lists convert them back into number of numpy arrays (converted again into lists)
      \item By iterating over lists collect each kmer that occured $occ_1$ times in train data class and $occ_2$ in the tested sample
      \item Calculate the contribution to the overall score:
      \begin{enumerate}
        \scriptsize
        \item For $occ_1 > 0$ and $occ_2 > 0$ we set $points \pluseq len(bin(min(occ_1, occ_2))); points_all \pluseq len(bin(max(occ_1, occ_2)))$
        \item For $occ_1 > 0$ and $occ_2 == 0$ we set $points_{all} \pluseq len(bin(occ_1))$
        \item For $occ_1 == 0$ and $occ_2 > 0$ we set $points_{all} \pluseq len(bin(occ_2))$
      \end{enumerate}
      \item Similarity score is then incremented by: $score \pluseq points / points_{all}$
    \end{enumerate}
  \end{enumerate}
\end{frame}
