macro(find_Python3_package PACKAGE VARIABLE)
  if(NOT DEFINED "${VARIABLE}")
    if(NOT CMAKE_REQUIRED_QUIET)
      # CMake 3.17+ use CHECK_START/CHECK_PASS/CHECK_FAIL
      message(STATUS "Looking for ${PACKAGE}")
    endif()
    if(Python3_VERSION VERSION_GREATER_EQUAL "3.8.0")
      # importlib.metadata was added in version 3.8
      set(CMD "from importlib.metadata import version; print(version('${PACKAGE}'), end='')")
      execute_process(
        COMMAND ${Python3_EXECUTABLE} -c "${CMD}"
        RESULT_VARIABLE EXIT_CODE
        OUTPUT_VARIABLE ${PACKAGE}_VERSION
        ERROR_QUIET
      )
    else()
      # importlib_metadata backport needed for older Python3 versions
      execute_process(
        COMMAND ${Python3_EXECUTABLE} -c "import importlib_metadata"
        RESULT_VARIABLE EXIT_CODE
        OUTPUT_QUIET ERROR_QUIET
      )
      if(EXIT_CODE EQUAL 0)
        set(CMD "from importlib_metadata import version; print(version('${PACKAGE}'), end='')")
        execute_process(
            COMMAND ${Python3_EXECUTABLE} -c "${CMD}"
            RESULT_VARIABLE EXIT_CODE
            OUTPUT_VARIABLE ${PACKAGE}_VERSION
            ERROR_QUIET
        )
      else()
        message(WARNING "importlib_metadata backport package required for Python <3.8")
      endif()
    endif()
    if(EXIT_CODE EQUAL 0)
      if(NOT CMAKE_REQUIRED_QUIET)
        message(STATUS "Looking for ${PACKAGE} - found version \"${${PACKAGE}_VERSION}\"")
      endif()
      set(${VARIABLE} 1 CACHE INTERNAL "Have package ${PACKAGE}")
    else()
      if(NOT CMAKE_REQUIRED_QUIET)
        message(STATUS "Looking for ${PACKAGE} - not found")
      endif()
      set(${VARIABLE} "" CACHE INTERNAL "Have package ${PACKAGE}")
    endif()
  endif()
endmacro()

			OUTPUT "${JOB_OUT_DIRECTORY_ASSEMBLY}/vars.tex"
			COMMAND ${PY_INTERPRETER} ${VARS_CONFIGURE_PATH} mode-tex ${JOB_OUT_DIRECTORY_ASSEMBLY}/vars.tex ${VARS_META_PATH} "${JOB_OUT_DIRECTORY_ASSEMBLY}/main.tex" "draft=false" "hasbibliography=false" "docepilogue=${DOC_EPILOGUE}" "docpreamble=${DOC_PREAMBLE}" ${COMMON_VARS_SPECS}
			DEPENDS ${VENV_DEPENDENCIES} "${JOB_OUT_DIRECTORY_ASSEMBLY}/main.tex" ${VARS_CONFIGURE_PATH} ${VARS_IMPLEMENTATION_PATH} ${VARS_META_PATH}
			COMMENT "${TEX_JOB}: Generate assembly Latex variables"
		)

		ADD_CUSTOM_COMMAND(
			OUTPUT ${JOB_OUT_DIRECTORY_ASSEMBLY}/main.aux ${JOB_OUT_DIRECTORY_ASSEMBLY}/main.pdf
			COMMAND ${LATEX_COMPILER_WRAP} --times=2 "${JOB_OUT_DIRECTORY_ASSEMBLY}"
			WORKING_DIRECTORY ${JOB_OUT_DIRECTORY_ASSEMBLY}
			DEPENDS "${JOB_OUT_DIRECTORY_ASSEMBLY}/main.tex" "${JOB_OUT_DIRECTORY_ASSEMBLY}/vars.tex"
			COMMENT "${TEX_JOB}: Assembly final resources"
		)
	ENDIF()

	ADD_CUSTOM_COMMAND(
		OUTPUT ${OUT_DIRECTORY_PDF}/${TEX_JOB}.pdf
		COMMAND ${MKDIR_COMMAND} ${OUT_DIRECTORY_PDF}
		COMMAND cp -f ${JOB_OUT_DIRECTORY_ASSEMBLY}/main.pdf ${OUT_DIRECTORY_PDF}/${TEX_JOB}.pdf
		COMMAND echo "[OK] Generated ${TEX_JOB}.pdf"
		WORKING_DIRECTORY ${JOB_OUT_DIRECTORY_ASSEMBLY}
		DEPENDS "${JOB_OUT_DIRECTORY_ASSEMBLY}/main.pdf"
		COMMENT "${TEX_JOB}: Export artifacts"
	)

	ADD_CUSTOM_COMMAND(
		OUTPUT "${JOB_OUT_DIRECTORY_TMP}/spellcheck.txt"
		COMMAND ${PY_INTERPRETER} ${VARS_CONFIGURE_PATH} mode-ltex-json ${JOB_OUT_DIRECTORY_TMP}/ltex.config.json ${VARS_META_PATH} "${JOB_OUT_DIRECTORY_ASSEMBLY}/main.tex" "draft=false" "hasbibliography=false" "docepilogue=${DOC_EPILOGUE}" "docpreamble=${DOC_PREAMBLE}" ${COMMON_VARS_SPECS}
		COMMAND ${LTEX_CMD} --client-configuration=${JOB_OUT_DIRECTORY_TMP}/ltex.config.json ${JOB_OUT_DIRECTORY_ASSEMBLY}/main.tex 2> ${JOB_OUT_DIRECTORY_TMP}/spellcheck.txt
		WORKING_DIRECTORY ${JOB_OUT_DIRECTORY_ASSEMBLY}
		BYPRODUCTS "${JOB_OUT_DIRECTORY_TMP}/ltex.config.json"
		DEPENDS "${JOB_OUT_DIRECTORY_ASSEMBLY}/main.tex"
		COMMENT "${TEX_JOB}: Spell checking"
	)


	LIST(APPEND LATEX_ALL_TARGETS "${OUT_DIRECTORY_PDF}/${TEX_JOB}.pdf")
	LIST(APPEND LATEX_ALL_TARGETS_CLEAN "clean-${TEX_JOB}")

	ADD_CUSTOM_TARGET("${TEX_JOB}"
		DEPENDS "${OUT_DIRECTORY_PDF}/${TEX_JOB}.pdf"
		COMMENT "Build Latex for ${TEX_JOB}"
	)

	ADD_CUSTOM_TARGET("spell-${TEX_JOB}"
		DEPENDS "${JOB_OUT_DIRECTORY_TMP}/spellcheck.txt"
		COMMENT "Spell check Latex for ${TEX_JOB}"
	)

	ADD_CUSTOM_TARGET("clean-${TEX_JOB}"
		COMMAND rm -rfd "${OUT_DIRECTORY_PDF}/${TEX_JOB}.pdf" "${CMAKE_SOURCE_DIR}/${LATEX_OUTPUT_PATH}/${TEX_JOB}"
		COMMENT "Spell check Latex for ${TEX_JOB}"
	)

	FOREACH(EXPORTER ${JOB_USED_EXPORTERS})
		SET( JOB_OUT_DIRECTORY_EXPORTER "${CMAKE_SOURCE_DIR}/${LATEX_OUTPUT_PATH}/${TEX_JOB}/_${EXPORTER}" )
		SET("EXPORTER_DEPS" ${EXPORTER_DEPS_${EXPORTER}})
		INCLUDE("${LATEX_EXPORTERS_DIR}/${EXPORTER}/CMakeLists.txt")
	ENDFOREACH()
	

ENDFOREACH()

ADD_CUSTOM_TARGET("latex-clean"
	DEPENDS ${LATEX_ALL_TARGETS_CLEAN}
	COMMENT "Clean all resources"
)

ADD_CUSTOM_TARGET("latex-all"
	DEPENDS ${LATEX_ALL_TARGETS}
	COMMENT "Build all Latex resources"
)

ADD_CUSTOM_TARGET(all-formats ALL) # Entry point of execution.
ADD_DEPENDENCIES( all-formats "latex-all")
