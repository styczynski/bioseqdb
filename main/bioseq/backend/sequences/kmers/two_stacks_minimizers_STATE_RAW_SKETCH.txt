// =================================================================================
// STATE_RAW — sketch / structural draft (Phase-3 report §8 follow-up,
// REPORT_PHASE_3_INDEPTH_ANALYSIS.md §7.2).
//
// Rationale: cross-iteration state variables (state_fwd, state_rc,
// prefix_min, prefix_kmer, pos_local, last_emit_pos) are bundled in
// `vec_stream<D, ST, NumStreams>` in STDARR. Phase-3's recommended
// follow-up moves them out of the bundle into raw `std::array<vec_t,
// NumStreams>` — keeping per-BP temporaries in `vec_stream` because
// their lifetime is bounded.
//
// The advantage: clang's mem2reg can promote `vec_t arr[N]` slots
// independently, just like the named-scalar Phase-1 layout, while the
// per-BP temporaries (which never escape the lambda body) still benefit
// from the `vec_stream` operator/value semantics.
//
// THIS IS A SKETCH ONLY. Full implementation requires ~30 LOC of edits
// to `two_stacks_minimizers.cppm`'s lambda body — every state-touching
// expression becomes a `for_each_index<NumStreams>` block.
// =================================================================================

// State declarations change:
//
// Before (STDARR):
//
//     vec_stream_t state_fwd      = vec_stream_t::splat(d, static_cast<storage_unit>(0));
//     vec_stream_t state_rc       = vec_stream_t::splat(d, static_cast<storage_unit>(0));
//     vec_stream_t prefix_min     = vec_stream_t::splat(d, max_val);
//     vec_stream_t prefix_kmer    = vec_stream_t::splat(d, static_cast<storage_unit>(0));
//     vec_stream_t pos_local      = vec_stream_t::splat(d, static_cast<storage_unit>(0));
//     vec_stream_t last_emit_pos  = vec_stream_t::splat(d, max_val);
//
// After (STATE_RAW):
//
//     std::array<vec_t, NumStreams> state_fwd;
//     std::array<vec_t, NumStreams> state_rc;
//     std::array<vec_t, NumStreams> prefix_min;
//     std::array<vec_t, NumStreams> prefix_kmer;
//     std::array<vec_t, NumStreams> pos_local;
//     std::array<vec_t, NumStreams> last_emit_pos;
//     for_each_index<NumStreams>([&](auto s_const) [[gnu::always_inline]] {
//         constexpr size_t s = decltype(s_const)::value;
//         state_fwd[s]     = hn::Zero(d);
//         state_rc[s]      = hn::Zero(d);
//         prefix_min[s]    = v_max;
//         prefix_kmer[s]   = hn::Zero(d);
//         pos_local[s]     = hn::Zero(d);
//         last_emit_pos[s] = v_max;
//     });
//
// Per-BP body changes example — forward / RC update:
//
// Before (STDARR):
//
//     state_fwd = And(Or(ShiftLeft<bpc>(state_fwd), bp_v), v_kmer_mask);
//
// After (STATE_RAW):
//
//     for_each_index<NumStreams>([&](auto s_const) [[gnu::always_inline]] {
//         constexpr size_t s = decltype(s_const)::value;
//         state_fwd[s] = hn::And(
//             hn::Or(hn::ShiftLeft<seq_traits::bits_per_char>(state_fwd[s]),
//                    bp_v[s]),
//             v_kmer_mask);
//     });
//
// (Note: `bp_v` remains a `vec_stream_t` because it's a per-BP temporary —
// only state escapes the lambda body and so only state needs to be raw.)
//
// Per-BP body changes example — Min/Lt for lex/strand:
//
// Before (STDARR):
//
//     const vec_stream_t  lex    = Min(state_fwd, state_rc);
//     const mask_stream_t strand = Lt(state_rc, state_fwd);
//
// After (STATE_RAW): bundle the result into a vec_stream_t manually:
//
//     const auto lex_pack = [&]<size_t... S>(std::index_sequence<S...>) {
//         return vec_stream_t::pack(hn::Min(state_fwd[S], state_rc[S])...);
//     }(std::make_index_sequence<NumStreams>{});
//     // (lex_pack is a vec_stream_t; rest of the body uses it by value)
//
// Or:
//
//     vec_stream_t lex;
//     for_each_index<NumStreams>([&](auto s_const) [[gnu::always_inline]] {
//         constexpr size_t s = decltype(s_const)::value;
//         lex[s] = hn::Min(state_fwd[s], state_rc[s]);
//     });
//
// (The `pack(...)` form is preferred because it keeps the single-point-
//  of-construction property `vec_stream::pack` was designed for.)
//
// EXPECTED OUTCOME: clang's mem2reg promotes state[s] slots like the
// Phase-1 named-scalar layout did. The lambda STILL doesn't fully inline
// (the §6 mechanism is not addressed here), so the residual cost vs
// DIRECT remains. But the *state-access* part of the cost in §5.1 is
// eliminated.
//
// Empirically the Phase-3 report estimated this would recover ~5-10 % of
// the regression. Combined with LAMBDAFREE (sketch_2.txt below), the
// expectation is full parity with DIRECT.
