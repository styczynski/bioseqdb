namespace bioseq::simd::HWY_NAMESPACE::two_stack_internal {

template<
    bool   DetectPalindrome,
    bool   IsFull,
    bool   HasStaticW,
    bool   HasStaticK,
    size_t StaticW,
    size_t StaticK,
    size_t NumStreams,
    size_t BufCapacity,
    class  D,
    typename ST,
    typename SeqTraits,
    typename OffsetsArray,
    template<typename H_A> typename simd_hash_t,
    class  AlphabetT,
    class  MinimizerT,
    typename HashValT
>
[[gnu::always_inline]] HWY_INLINE void
step_one_bp_vec_stream(
    // === per-call values ===
    D                                      d,
    vec_stream<D, ST, NumStreams>          bp_v,           // pass by VALUE (HVA on M1 NEON)
    hn::Mask<D>                            live_mask,
    OffsetsArray const&                    offsets,
    // === cross-iteration state by REFERENCE ===
    vec_stream<D, ST, NumStreams>&         state_fwd,
    vec_stream<D, ST, NumStreams>&         state_rc,
    vec_stream<D, ST, NumStreams>&         prefix_min,
    vec_stream<D, ST, NumStreams>&         prefix_kmer,
    vec_stream<D, ST, NumStreams>&         pos_local,
    vec_stream<D, ST, NumStreams>&         last_emit_pos,
    // === ring buffers by REFERENCE ===
    std::array<vec_stream<D, ST, NumStreams>, BufCapacity>& ring_buf_elem,
    std::array<vec_stream<D, ST, NumStreams>, BufCapacity>& ring_buf_hash_strand,
    // === scalar bookkeeping by REFERENCE ===
    ST&                                    pos_wrap_delta,
    size_t&                                step,
    size_t&                                ring_idx,
    size_t&                                next_wrap_step,
    // === loop-invariant SIMD constants by VALUE ===
    hn::Vec<D>                             v_kmer_mask,
    hn::Vec<D>                             v_alphabet_complement_xor,
    hn::Vec<D>                             v_max,
    hn::Vec<D>                             v_one,
    hn::Vec<D>                             v_pos_mask,
    hn::Vec<D>                             v_hash_kmer_mask,
    hn::Vec<D>                             v_strand_clear,
    hn::Vec<D>                             v_golden,
    hn::Vec<D>                             v_wrap,
    // === loop-invariant scalars by VALUE ===
    size_t                                 w,
    size_t                                 k_clamped,
    size_t                                 emit_warmup,
    size_t                                 wrap_period,
    HashValT                               kmer_mask_scalar,
    ST                                     wrap_delta_amount,
    // === collector by REFERENCE ===
    sketch_collector_t<MinimizerT>&        collector
) {
    using vec_t   = hn::Vec<D>;
    using mask_t  = hn::Mask<D>;

    static_assert(/* place sanity static_asserts here, e.g. NumStreams ∈ {1,2} */ true);

    // ---- BODY: copy verbatim from the existing step_one_bp lambda in
    //      two_stacks_minimizers.cppm (lines 285..481), substituting the
    //      capture references with the named parameters above. The
    //      semantic body does not change — only the syntactic capture
    //      mechanism. ----

    // Forward / RC update
    const vec_stream<D, ST, NumStreams> complemented = Xor(bp_v, v_alphabet_complement_xor);
    state_fwd = And(Or(ShiftLeft<SeqTraits::bits_per_char>(state_fwd), bp_v), v_kmer_mask);

    vec_stream<D, ST, NumStreams> new_high;
    if constexpr (HasStaticK) {
        constexpr size_t static_rev_shift = SeqTraits::bits_per_char * (StaticK - 1);
        constexpr size_t storage_bits_max = sizeof(ST) * 8;
        constexpr size_t clamped_rev_shift =
            (static_rev_shift >= storage_bits_max) ? (storage_bits_max - 1) : static_rev_shift;
        new_high = ShiftLeft<clamped_rev_shift>(complemented);
    } else {
        const size_t dynamic_rev_shift = SeqTraits::bits_per_char * (k_clamped - 1);
        new_high = ShiftLeftSame(complemented, static_cast<int>(dynamic_rev_shift));
    }
    state_rc = Or(ShiftRight<SeqTraits::bits_per_char>(state_rc), new_high);

    // Lex + strand classification
    const auto lex    = Min(state_fwd, state_rc);
    const auto strand = Lt(state_rc, state_fwd);

    // Mixed hash, packed element, ring-buf write, prefix-min update,
    // suffix-min rebuild every w, position wrap, emit — all verbatim from
    // existing lambda body.
    //
    // (~200 LOC continues; omitted here; see two_stacks_minimizers_STDARR.txt
    //  lines 285..481 for the complete body and translate captures→params.)
    //
    // For the static suffix-min rebuild and wrap-slot, the inner lambdas
    // can stay (they take size_t indices via integer_sequence and don't
    // suffer from outer-lambda capture pressure).
}

}  // namespace bioseq::simd::HWY_NAMESPACE::two_stack_internal
